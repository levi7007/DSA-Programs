21.Program to Print a New Line in String (Each word in newline)


//PRINT ALL THE PERMUTATION OF A STRING I.E. "ABCDE".
#include<stdio.h>
#include<string.h>
void swap(char *a,char *b){
    char t=*a;
    *a=*b;
    *b=t;
}
void perm(char s[],int l,int r){
    int i;
    if(l==r){
        printf("%s\n",s);
    }
    else{
        for(i=l;i<=r;i++){
            swap(&s[l],&s[i]);
            perm(s,l+1,r);
            swap(&s[l],&s[i]);
        }
    }
}
int main(){
    char s[50];
    printf("enter the string: ");
    gets(s);
    perm(s,0,strlen(s)-1);
    return 0;
}

ABCDE

ABDCE
