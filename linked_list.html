1.IMPLEMENT THE OPERATIONS OF SINGLE LINKED LIST STORING INTEGER VALUES (INSERT AT BEGIN,INSERT AT POS,INSERT AT END,DELETE AT BEGIN,DELETE AT END, DELETE AT POS, ISEMPTY, ISFULL, DISPLAY, SEARCH) 


#include<stdio.h>
#include<stdlib.h>
struct node{
    int data;
    struct node *n;
};
struct node *start=NULL;

int input(){
    int key;
    printf("\n-----MENU-----\n");
    printf("1.Insert at begin\n2.Insert at position\n3.Insert at end\n");
    printf("4.Delete at begin\n5.Delete at end\n6.Delete at position\n");
    printf("7.IsEmpty\n8.IsFull\n9.Display\n10.Search\n11.Exit\n");
    scanf("%d",&key);
    return key;
}

void insert_begin(int x){
    struct node *t;
    t=(struct node*)malloc(sizeof(struct node));
    if(t==NULL){
        printf("List is full\n");
        return;
    }
    t->data=x;
    t->n=start;
    start=t;
}

void insert_end(int x){
    struct node *t,*p;
    t=(struct node*)malloc(sizeof(struct node));
    if(t==NULL){
        printf("List is full\n");
        return;
    }
    t->data=x;
    t->n=NULL;
    if(start==NULL){
        start=t;
        return;
    }
    p=start;
    while(p->n!=NULL)
        p=p->n;
    p->n=t;
}

void insert_pos(int x,int pos){
    int i;
    struct node *t,*p;
    if(pos<1){
        printf("Invalid position\n");
        return;
    }
    if(pos==1){
        insert_begin(x);
        return;
    }
    p=start;
    for(i=1;i<pos-1&&p!=NULL;i++)
        p=p->n;
    if(p==NULL){
        printf("Invalid position\n");
        return;
    }
    t=(struct node*)malloc(sizeof(struct node));
    if(t==NULL){
        printf("List is full\n");
        return;
    }
    t->data=x;
    t->n=p->n;
    p->n=t;
}

void delete_begin(){
    struct node *t;
    if(start==NULL){
        printf("List is empty\n");
        return;
    }
    t=start;
    start=start->n;
    free(t);
    printf("Node deleted\n");
}

void delete_end(){
    struct node *t,*p;
    if(start==NULL){
        printf("List is empty\n");
        return;
    }
    t=start;
    if(start->n==NULL){
        start=NULL;
        free(t);

}
