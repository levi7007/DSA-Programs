7.Create a tic tac toe game using 2D Arrays.


#include <stdio.h>
char mat[3][3]={{' ',' ',' '},{' ',' ',' '},{' ',' ',' '}};
char win(char p){
    for(int i=0;i<3;i++){
        if(mat[i][0]==p&&mat[i][1]==p&&mat[i][2]==p){
            printf("winner %c\nhahhhhh loser\n",p);
            return p;
        }
        else if(mat[0][i]==p&&mat[1][i]==p&&mat[2][i]==p){
            printf("winner %c\nhahhhhh loser\n",p);
            return p;
        }
    }
    if(mat[0][0]==p&&mat[1][1]==p&&mat[2][2]==p||mat[0][2]==p&&mat[1][1]==p&&mat[2][0]==p){
        printf("winner %c\nhahhhhh loser\n",p);
        return p;
    }
    return ' ';
}
int main(){
    int r,c,count=0;
    char x,o;
    printf("welcome to the tic-tac-toe game!\n");
    for(int i=0;i<3;i++){
        for(int j=0;j<3;j++){
            printf("enter the row: ");
            scanf("%d",&r);
            printf("enter the column: ");
            scanf("%d",&c);
            if(r<1||r>3||c<1||c>3){
                printf("invalid\n");
                j--;
                continue;
            }
            if(mat[r-1][c-1]!=' '){
                printf("already filled\n");
                j--;
                continue;
            }
            count++;
            if((count%2)!=0){
                mat[r-1][c-1]='X';
            }
            else if(count%2==0){
                mat[r-1][c-1]='O';
            }
            for(int i=0;i<3;i++){
                for(int j=0;j<3;j++){
                    printf("[%c]",mat[i][j]);
                }
                printf("\n");
            }
            if(count>4){
                x=win('X');
                o=win('O');
                if(x=='X'||o=='O'){
                    return 0;
                }
            }
        }
    }
    printf("draw !!\n");
    return 0;
}
