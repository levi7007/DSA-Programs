4.CREATE THE GRAPH AS A ADJACENCY LIST WITH OPERATIONS SUCH AS INSERTION, DELETION 


#include<stdio.h>
#include<stdlib.h>
struct node{
    int data;
    struct node *n;
};
struct node *g[3]={NULL};

int input(){
    int key;
    printf("\n-----MENU-----\n");
    printf("1.insert\n2.delete\n3.display\n4.exit\n");
    scanf("%d",&key);
    return key;
}

void insert_edge(int s,int d){
    struct node *t;
    if(s>=1&&s<=3&&d>=1&&d<=3){
        t=(struct node*)malloc(sizeof(struct node));
        t->data=d;
        t->n=g[s-1];
        g[s-1]=t;
    }
    else{
        printf("invalid source/destination\n");
    }
}

void delete_edge(int s,int d){
    struct node *t,*p;
    if(s<1||s>3||d<1||d>3){
        printf("invalid source/destination\n");
        return;
    }
    t=g[s-1];
    p=NULL;
    while(t!=NULL){
        if(t->data==d){
            if(p==NULL)
            g[s-1]=t->n;
            else
            p->n=t->n;
            free(t);
            printf("edge deleted\n");
            return;
        }
        p=t;
        t=t->n;
    }
    printf("edge not found\n");
}

void display(){
    int i;
    struct node *t;
    printf("Adjacency List:\n");
    for(i=0;i<3;i++){
        printf("%d -> ",i+1);
        t=g[i];
        while(t!=NULL){
            printf("%d -> ",t->data);
            t=t->n;
        }
        printf("NULL\n");
    }
}
