void ms(int a[]){
    int b[5],i,j,k,s,m,e,c=0;
    for(s=1;s<5;s=s*2){
        for(i=0;i<5-s;i=i+2*s){
            m=i+s-1;
            e=i+2*s-1;
            if(e>=5){
                e=4;
            }
            j=i;
            k=m+1;
            int p=i;
            while(j<=m && k<=e){
                c++;
                if(a[j]<a[k]){
                    b[p++]=a[j++];
                }
                else{
                    b[p++]=a[k++];
                }
            }
            while(j<=m){
                b[p++]=a[j++];
            }
            while(k<=e){
                b[p++]=a[k++];

                a[j]=b[j];
            }
        }

    i=l;
    j=m+1;
    k=l;

void qs(int a[]){
    int stack[100],top=-1;
    int l=0,r=4,i,j,pivot,temp;
    stack[++top]=l;
    stack[++top]=r;
    while(top>=0){
        r=stack[top--];
        l=stack[top--];
        pivot=a[r];
        i=l-1;
        for(j=l;j<r;j++){
            if(a[j]<pivot){
                i++;
                temp=a[i];
                a[i]=a[j];
                a[j]=temp;
            }
        }
        temp=a[i+1];
        a[i+1]=a[r];
        a[r]=temp;
        int p=i+1;
        if(p-1>l){
            stack[++top]=l;
            stack[++top]=p-1;
        }
        if(p+1<r){
            stack[++top]=p+1;
            stack[++top]=r;
        }
    }
}

int partition(int a[],int l,int r,int *c){
    int pivot=a[r],i=l-1,j,temp;
    for(j=l;j<r;j++){
        (*c)++;
        if(a[j]<pivot){
            i++;
            temp=a[i];
            a[i]=a[j];
            a[j]=temp;
        }
    }
    temp=a[i+1];
    a[i+1]=a[r];
    a[r]=temp;
    return i+1;
}
void qs(int a[],int l,int r,int *c){
    int p;
    if(l<r){
        p=partition(a,l,r,c);
        qs(a,l,p-1,c);
        qs(a,p+1,r,c);
    }
}

void merge(char a[][11],char b[][11],int l,int m,int r,int *c){
    int i=l,j=m+1,k=l;
    while(i<=m && j<=r){
        (*c)++;
        if(strcmp(a[i],a[j])<=0){
            strcpy(b[k],a[i]);
            i++;
        }
        else{
            strcpy(b[k],a[j]);
            j++;
        }
        k++;
    }
    while(i<=m){
        strcpy(b[k],a[i]);
        i++;
        k++;
    }
    while(j<=r){
        strcpy(b[k],a[j]);
        j++;
        k++;
    }
    for(i=l;i<=r;i++){
        strcpy(a[i],b[i]);
    }
}
void ms(char a[][11],char b[][11],int n,int *c){
    int size,l,m,r;
    for(size=1;size<n;size=size*2){
        for(l=0;l<n-1;l=l+2*size){
            m=l+size-1;
            r=l+2*size-1;
            if(m>=n){
                continue;
            }
            if(r>=n){
                r=n-1;
            }
            merge(a,b,l,m,r,c);
        }
    }
}

    int l,h,p;
