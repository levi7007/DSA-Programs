6.ENTER A STRING FROM THE USER, ENTER A CHARACTER AND PERFORM BINARY SEARCH ALGORITHM THROUGH RECURSION. 


#include<stdio.h>
#include<string.h>

void input(char s[]){
printf("enter a string: ");
gets(s);
}

char search(){
char key;
printf("enter the character to search: ");
scanf("%c",&key);
return key;
}

void sort(char s[]){
int i,j;
char t;
for(i=0;s[i]!='\0';i++){
for(j=i+1;s[j]!='\0';j++){
if(s[i]>s[j]){
t=s[i];
s[i]=s[j];
s[j]=t;
}
}
}
}

int bs(char s[],char x,int l,int r){
int mid;
if(l>r){
return 0;
}
mid=(l+r)/2;
if(s[mid]==x){
return 1;
}
else if(x<s[mid]){
return bs(s,x,l,mid-1);
}
else{
return bs(s,x,mid+1,r);
}
}

int main(){
char s[50],x;
input(s);
sort(s);
printf("sorted string: %s\n",s);
x=search();
if(bs(s,x,0,strlen(s)-1)==1){
printf("character found");
}
else{
printf("character not found");
}
return 0;
}

Output:

enter a string: COMPUTER
sorted string: CEMOPRTU
enter the character to search: P
character found

10.CALCULATE THE BEST AND WORST COMPLEXITY OF THE BINARY SEARCHING ALGORITHM WITH EXACT TIME TAKEN BY THE ALGORITHM ON INTEGER ARRAY. 


#include<stdio.h>

void input(int a[],int n){
int i;
for(i=0;i<n;i++){
printf("enter array element: ");
scanf("%d",&a[i]);
}
}

void sort(int a[],int n){
int i,j,temp;
for(i=0;i<n-1;i++){
for(j=0;j<n-i-1;j++){
if(a[j]>a[j+1]){
temp=a[j];
a[j]=a[j+1];
a[j+1]=temp;
}
}
}
}

void bs(int a[],int n,int x){
int l=0,r=n-1,mid,f=0,c=0;
while(l<=r){
mid=(l+r)/2;
c++;
if(a[mid]==x){
f=1;
break;
}
else if(a[mid]<x){
l=mid+1;
}
else{
r=mid-1;
}
}
if(f==1){
printf("\nelement found at position %d",mid+1);
}
else{
printf("\nelement not found");
}
printf("\nnumber of comparisons: %d",c);
printf("\n\nbest case complexity: O(1)");
printf("\nworst case complexity: O(log n)");
printf("\naverage case complexity: O(log n)");
}

int main(){

printf("\nsorted array:\n");
for(i=0;i<n;i++){
printf("%d ",a[i]);
}
printf("\nenter element to search: ");
scanf("%d",&x);
bs(a,n,x);
return 0;
}
