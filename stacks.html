8.SORT THE STACK IN ASCENDING ORDER USING INTEGER ARRAY. 


#include<stdio.h>
int arr[5];
int top=-1;
void push(int x){
    if(top==4){
        printf("overflow\n");
    }
    else{
        top++;
        arr[top]=x;
    }
}
void sort(){
    int i,j,temp;
    for(i=0;i<=top;i++){
        for(j=i+1;j<=top;j++){
            if(arr[i]<arr[j]){
                temp=arr[i];
                arr[i]=arr[j];
                arr[j]=temp;
            }
        }
    }
}

void display(){
    int i;
    if(top==-1){
        printf("stack is empty\n");
    }
    else{
        printf("stack elements: ");
        for(i=top;i>=0;i--){
            printf("%d ",arr[i]);
        }
        printf("\n");
    }
}

    int key,x;
    do{
        printf("\n-----MENU-----\n");
        printf("1.push\n2.sort\n3.display\n4.exit\n");
        scanf("%d",&key);
        if(key==1){
            printf("enter element: ");
            scanf("%d",&x);
            push(x);
        }
        else if(key==2){
            sort();
            printf("stack sorted\n");

        }

            printf("exit\n");
        }

            printf("invalid\n");

   return 0;

9.ENTER THE INFIX EXPRESSION FROM USER AND CONVERT INTO PREFIX EXPRESSION. 


#include<stdio.h>
#include<string.h>
char stack[50];
int top=-1;

void push(char x){
    top++;
    stack[top]=x;
}

char pop(){
    char x=stack[top];
    top--;
    return x;
}

int priority(char x){
    if(x=='^'){
        return 3;
    }
    else if(x=='*'||x=='/'||x=='%'){
        return 2;
    }
    else if(x=='+'||x=='-'){
        return 1;
    }
    else{
        return 0;
    }
}

void reverse(char str[]){
    int i,j;
    char x;
    i=0;
    j=strlen(str)-1;
    while(i<j){
        x=str[i];
        str[i]=str[j];
        str[j]=x;
        i++;
        j--;
    }
}

int main(){
    char infix[50],prefix[50],x;
    int i,j=0;
    printf("enter infix expression: ");
    scanf("%s",infix);
    reverse(infix);
    for(i=0;infix[i]!='\0';i++){
        if(infix[i]=='('){
            infix[i]=')';
        }
        else if(infix[i]==')'){
            infix[i]='(';
        }
    }
    for(i=0;infix[i]!='\0';i++){
        x=infix[i];
        if((x>='A'&&x<='Z')||
           (x>='a'&&x<='z')||
           (x>='0'&&x<='9')){
            prefix[j]=x;
            j++;
        }
        else if(x=='('){
            push(x);
        }
        else if(x==')'){
            while(top!=-1 && stack[top]!='('){
                prefix[j]=pop();
                j++;
            }
            if(top!=-1){
                pop();
            }
        }
        else{
            while(top!=-1 && stack[top]!='(' &&
                  priority(stack[top])>priority(x)){
                prefix[j]=pop();
                j++;
            }
            push(x);
        }
    }
    while(top!=-1){
        prefix[j]=pop();
        j++;
    }
    prefix[j]='\0';
    reverse(prefix);
    printf("prefix expression: %s\n",prefix);

    return 0;
}

                postfix[j]=pop();
                j++;
            }
            if(top!=-1){
                pop();
            }
        }
        else{
            while(top!=-1 && stack[top]!='(' &&
                  (priority(stack[top])>priority(x) ||
                  (priority(stack[top])==priority(x)&&x!='^'))){
                postfix[j]=pop();
                j++;
            }
            push(x);
        }
    }

    while(top!=-1){
        postfix[j]=pop();
        j++;
    }
    postfix[j]='\0';
    printf("postfix expression: %s\n",postfix);

    return x;

    for(i=strlen(eq)-1;i>=0;i--){
        if(eq[i]>='0'&&eq[i]<='9'){
            push(eq[i]-'0');

            a=pop();
            b=pop();

                x=a%b;
