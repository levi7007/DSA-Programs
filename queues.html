#include<stdio.h>
int q[5];
int front=-1,rear=-1;

int input(){
    int x;
    printf("----menu----\n");
    printf("1.enqueue\n2.dequeue\n3.peek\n4.isempty\n5.isfull\n6.exit\n");
    scanf("%d",&x);
    return x;
}

int enqueue(int x){
    if(rear==4){
        printf("queue is full\n");
        return 0;
    }
    else{
        if(front==-1){
            front=0;
        }
        rear++;
        q[rear]=x;
        return 1;
    }
}

int dequeue(){
    int x;
    if(front==-1){
        return 0;
    }
    else{
        x=q[front];
        front++;
        if(front>rear){
            front=-1;
            rear=-1;
        }
        return x;
    }
}

void peek(){
    if(front==-1){
        printf("queue is empty\n");
    }
    else{
        printf("front element is %d\n",q[front]);
    }
}

void isempty(){
    if(front==-1){
        printf("queue is empty\n");
    }
    else{
        printf("queue is not empty\n");
    }
}

void isfull(){
    if(rear==4){
        printf("queue is full\n");
    }
    else{
        printf("queue is not full\n");
    }
}

void display(){
    int i;
    if(front==-1){
        printf("queue is empty\n");
    }
    else{
        for(i=front;i<=rear;i++){
            printf("%d ",q[i]);
        }
        printf("\n");
    }
}

int main(){
    int key,x,a;
do{
    key=input();
    if(key==1){
        printf("enter data: ");
        scanf("%d",&x);
        enqueue(x);
    }
    else if(key==2){
        if(front==-1){
            printf("queue is empty\n");
        }
        else{
            a=dequeue();
            printf("deleted value is %d\n",a);
        }
    }
    else if(key==3){
        peek();

    return x;
}

    temp=(struct node *)malloc(sizeof(struct node));
    if(temp!=NULL){
        temp->data=x;
        temp->n=NULL;
        if(front==NULL){
            front=temp;
            rear=temp;
        }
        else{
            rear->n=temp;
            rear=temp;
        }
        return 1;
    }
    else{
        printf("error\n");
        return 0;
    }
}

int dequeue(){
    int x;

        return 0;

        x=temp->data;
        front=front->n;
        // if(front==NULL){

        // }
        free(temp);
        return x;
    }
}
