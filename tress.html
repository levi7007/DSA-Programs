#include<stdio.h>
#include<stdlib.h>
struct bst{
    int data;
    struct bst *left,*right;
};

struct bst *start=NULL;
struct bst *temp,*queue[100];
int front=-1,rear=-1;

struct bst* create(){
    int x;
    struct bst *temp;
    printf("enter data (-1 for no node): ");
    scanf("%d",&x);
    if(x==-1){
        return NULL;
    }
    temp=(struct bst*)malloc(sizeof(struct bst));
    temp->data=x;
    printf("enter left child of %d\n",x);
    temp->left=create();
    printf("enter right child of %d\n",x);
    temp->right=create();
    return temp;
}

void enqueue(struct bst *x){
    if(front==-1){
        front=0;
    }
    rear++;
    queue[rear]=x;
}

struct bst* dequeue(){
    struct bst *x;
    x=queue[front];
    if(front==rear){
        front=rear=-1;
    }
    else{
        front++;
    }
    return x;
}

void levelorder(struct bst *t){
    if(t==NULL){
        printf("tree is empty");
        return;
    }
    enqueue(t);
    printf("\nlevel order traversal: ");
    while(front!=-1){
        temp=dequeue();
        printf("%d ",temp->data);
        if(temp->left!=NULL){
            enqueue(temp->left);
        }
        if(temp->right!=NULL){
            enqueue(temp->right);
        }
    }
}

int main(){
    printf("create binary tree\n");
    start=create();
    levelorder(start);
    return 0;
}

    int i;
    struct bst *temp;
    if(s>e){
        return NULL;
    }
    temp=(struct bst*)malloc(sizeof(struct bst));
    temp->data=post[k--];
    temp->left=NULL;
    temp->right=NULL;
    if(s==e){
        return temp;
    }
    for(i=s;i<=e;i++){
        if(in[i]==temp->data){
            break;
        }
    }
    temp->right=create(post,in,i+1,e);
    temp->left=create(post,in,s,i-1);
    return temp;
}
void inorder(struct bst *t){
    if(t!=NULL){
        inorder(t->left);
        printf("%c ",t->data);
        inorder(t->right);
    }
}
void postorder(struct bst *t){
    if(t!=NULL){
        postorder(t->left);
        postorder(t->right);
        printf("%c ",t->data);
    }
}

void find(int x,struct bst *t){
    struct bst *p,*s;
    p=NULL;
    s=NULL;

    while(t!=NULL){
        if(x<t->data){
            s=t;
            t=t->left;
        }
        else if(x>t->data){
            p=t;
            t=t->right;
        }
        else{
            break;
        }
    }

        printf("element not found");
        return;
    }

    if(t->left!=NULL){

    temp->right=create();

        store(t->left);
        a[n++]=t->data;
        store(t->right);
    }

    for(i=0;i<n;i++){
        for(j=i+1;j<n;j++){
            if(a[i]>a[j]){

                a[i]=a[j];
                a[j]=t;
            }
        }
    }
}

        inorder(t->left);
        t->data=a[k++];
        inorder(t->right);
    }
}

        display(t->left);
        printf("%d ",t->data);
        display(t->right);
    }
}

#include<stdlib.h>

struct node{
int data;
struct node *left,*right;
};

struct node *start=NULL;
struct node *q[100];

int x;

printf("enter option: ");
scanf("%d",&x);

}

void create(int n){
int i,x,f,r;
struct node *temp;

f=0;
r=0;

for(i=0;i<n;i++){
printf("enter data: ");
scanf("%d",&x);

temp=(struct node*)malloc(sizeof(struct node));
temp->data=x;
temp->left=NULL;
temp->right=NULL;

if(start==NULL){
start=temp;
}
else{
if(q[f]->left==NULL){
q[f]->left=temp;
}
else{
q[f]->right=temp;
f++;
}
}
q[r++]=temp;
}
}

void preorder(struct node *t){
if(t!=NULL){
printf("%d ",t->data);
preorder(t->left);
preorder(t->right);
}
}

void inorder(struct node *t){
if(t!=NULL){
inorder(t->left);
printf("%d ",t->data);
inorder(t->right);
}
}

void postorder(struct node *t){
if(t!=NULL){
postorder(t->left);
postorder(t->right);
printf("%d ",t->data);
}
}

}
else if(key==2){
preorder(start);
}
else if(key==3){
inorder(start);
}
else if(key==4){

}
}while(key!=5);

}

struct bst *insert(int x,struct bst *t){
if(t==NULL){
temp=(struct bst*)malloc(sizeof(struct bst));
temp->data=x;
temp->left=NULL;
temp->right=NULL;
return temp;
}
if(x<t->data){
t->left=insert(x,t->left);
}
else if(x>t->data){
t->right=insert(x,t->right);
}
else{
printf("duplicate");
}
return t;
}

int count(struct bst *t){
if(t==NULL){
return 0;
}
return 1+count(t->left)+count(t->right);
}
