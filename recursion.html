1.CALCULATE FACTORIAL OF A NUMBER USING RECURSION 


#include<stdio.h>
int factorial(int x){
    if(x==1){
        return 1;
    }
    else{
    return x*factorial(x-1);
    }    
}

int main(){
int x,f;
printf("enter a number: ");
scanf("%d",&x);
f=factorial(x);
printf("factorial of %d is: %d",x,f);
    return 0;
}

Output:

enter a number: 5
factorial of 5 is: 120

2.CALCULATE FIBONACCI OF A NUMBER USING RECURSION 


#include<stdio.h>
int fib(int x){
    if(x<=1){
        return x;
    }
    return fib(x-1)+fib(x-2);
}

int main(){
int t;
printf("enter the number of terms: ");
scanf("%d",&t);
for(int i=0;i<t;i++){
    printf("%d ",fib(i));
}
    return 0;
}

Output:

enter the number of terms: 8
0 1 1 2 3 5 8 13

5.FIND THE LARGEST ELEMENT IN THE INTEGER ARRAY USING RECURSION. 


#include<stdio.h>
int largest(int x,int a[]){
    if(x==1){
        return a[0];
    }
    int l=largest(x-1,a);
    if(a[x-1]>l){
        return a[x-1];
    }
    else{
        return l;
    }
}

int main(){
    int x,l;
    printf("enter the size of an array: ");
    scanf("%d",&x);
    int arr[x];
    for(int i=0;i<x;i++){
        printf("enter element: ");
        scanf("%d",&arr[i]);
    }
    l=largest(x,arr);
    printf("largest element is: %d",l);
    return 0;
}

Output:

enter element: 12
enter element: 45
enter element: 23
enter element: 67
enter element: 34
largest element is: 67

7.FIND THE ARITHMETIC MEAN OF AN INTEGER ARRAY USING RECURSION. 


#include<stdio.h>
int AM(int a[],int x){
    if(x==0){
        return 0;
    }
    else{
        return a[x-1]+AM(a,x-1);
    }
}


int main(){
    int x;
    printf("enter the size of an array: ");
    scanf("%d",&x);
int arr[x];
int s;
float m;
for(int i=0;i<x;i++){
    printf("enter elements: ");
    scanf("%d",&arr[i]);
}
s=AM(arr,x);
m=s/x;
printf("Arithmetic Mean is: %.2f",m);

    return 0;
}

enter the size of an array: 3
enter elements: 2
enter elements: 4
